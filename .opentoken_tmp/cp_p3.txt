CP   241	    if (peer == null) return;
CP   242	    if (info.peerId != peer.id) {
CP   243	      _logger.w('Ignoring beacon from unknown peer: ${info.peerId}');
CP   244	      return;
CP   245	    }
CP   246	
CP   247	    _lastDiscoveredIp = info.ip;
CP   248	    _ref.read(connectionStatusProvider.notifier).recordBeacon(info.ip);
CP   249	
CP   250	    if (info.ip != peer.ip || info.tcpPort != peer.port) {
CP   251	      _logger.i('Peer discovered at new address ${info.ip}:${info.tcpPort}');
CP   252	      final updated = peer.copyWith(ip: info.ip, port: info.tcpPort);
CP   253	      _peer = updated;
CP   254	      _peerDao.update(updated);
CP   255	      _ref.read(peerProvider.notifier).applyUpdate(updated);
CP   256	      _ref.read(connectionStatusProvider.notifier).setPeerIp(info.ip);
CP   257	    }
CP   258	
CP   259	    if (!state && !_connecting) {
CP   260	      final verificationCode = (peer.key.hashCode.abs() % 1000000).toString().padLeft(6, '0');
CP   261	      _pairingRequestController.add(PairingRequest(
CP   262	        peerId: peer.id,
CP   263	        deviceName: info.deviceName,
CP   264	        ip: info.ip,
CP   265	        port: info.tcpPort,
CP   266	        verificationCode: verificationCode,
CP   267	      ));
CP   268	      _connectTo(info.ip, info.tcpPort);
CP   269	    }
CP   270	  }
