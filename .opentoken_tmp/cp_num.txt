206:  Future<void> _tryConnectToStoredPeer() async {
207:    final peer = _peer;
208:    final key = _key;
209:    if (peer == null || key == null) return;
210:    if (state || _connecting) return;

215:          .recordConnectAttempt('Eş cihaz IP bilinmiyor (beacon bekleniyor)');
216:      return;

222:  Future<bool> _connectTo(String ip, int port) async {
223:    final key = _key;
224:    if (key == null || _connecting || state) return false;
225:
226:    _connecting = true;
227:    try {
228:      _socketManager ??= _createSocketManager();
229:      final ok = await _socketManager!.connect(ip, port, key);
230:      _ref.read(connectionStatusProvider.notifier).recordConnectAttempt(
231:            ok ? null : 'Bağlantı başarısız: $ip:$port (sunucu kapalı veya ulaşılamıyor)',
232:          );
233:      return ok;
234:    } finally {
235:      _connecting = false;
236:    }

239:  void _onBeacon(BeaconInfo info) {
240:    final peer = _peer;
241:    if (peer == null) return;
242:    if (info.peerId != peer.id) {
243:      _logger.w('Ignoring beacon from unknown peer: ${info.peerId}');
244:      return;
245:    }
246:
247:    _lastDiscoveredIp = info.ip;
248:    _ref.read(connectionStatusProvider.notifier).recordBeacon(info.ip);
249:
250:    if (info.ip != peer.ip || info.tcpPort != peer.port) {
251:      _logger.i('Peer discovered at new address ${info.ip}:${info.tcpPort}');
252:      final updated = peer.copyWith(ip: info.ip, port: info.tcpPort);
253:      _peer = updated;
254:      _peerDao.update(updated);
255:      _ref.read(peerProvider.notifier).applyUpdate(updated);
256:      _ref.read(connectionStatusProvider.notifier).setPeerIp(info.ip);
257:    }
258:
259:    if (!state && !_connecting) {
260:      final verificationCode = (peer.key.hashCode.abs() % 1000000).toString().padLeft(6, '0');
261:      _pairingRequestController.add(PairingRequest(
262:        peerId: peer.id,
263:        deviceName: info.deviceName,
264:        ip: info.ip,
265:        port: info.tcpPort,
266:        verificationCode: verificationCode,
267:      ));
268:      _connectTo(info.ip, info.tcpPort);
269:    }
270:  }

345:  void _handleIncomingMessage(MirrorMessage message) async {
346:    final key = _key;
347:    if (key == null) return;
348:
349:    final decrypted = await CryptoManager.decrypt(key, message.payload);
350:    if (decrypted == null) {
351:      _logger.e('Decryption failed for message: ${message.id}');
352:      return;
353:    }
354:
355:    final payload = jsonDecode(decrypted) as Map<String, dynamic>;
356:    final now = DateTime.now();
357:
358:    switch (message.type) {
359:      case MessageTypes.callIncoming:
360:        final number = payload['number'] as String? ?? 'unknown';
361:        final id = payload['id'] as String? ?? message.id;
362:        await _ref.read(callListProvider.notifier).add(CallEvent(
363:              id: id,
364:              direction: 'incoming',
365:              number: number,
366:              timestamp: DateTime.fromMillisecondsSinceEpoch(
367:                  payload['timestamp'] as int? ?? now.millisecondsSinceEpoch),
368:              encrypted: message.payload,
369:              status: 'delivered',
370:              createdAt: now,
371:            ));
372:        await _notify(
373:          id: int.tryParse(id) ?? 1,
374:          title: 'Gelen Arama',
375:          body: number,
376:          payload: message.id,
377:        );
378:        break;
379:
380:      case MessageTypes.callRejected:
381:        if (isSource) {
382:          await TelephonyChannel.rejectCall();
383:        }
384:        final id = payload['id'] as String?;
385:        if (id != null) {
386:          await _ref.read(callListProvider.notifier).updateStatus(id, 'rejected');
387:        }
388:        break;
389:
390:      case MessageTypes.smsIncoming:
391:        final address = payload['address'] as String? ?? 'unknown';
392:        final body = payload['body'] as String? ?? '';
393:        final id = payload['id'] as String? ?? message.id;
394:        await _ref.read(smsListProvider.notifier).add(SmsMessage(
395:              id: id,
396:              threadId: payload['thread_id'] as String? ?? '',
397:              address: address,
398:              body: body,
399:              encrypted: message.payload,
400:              direction: 'incoming',
401:              status: 'received',
402:              timestamp: DateTime.fromMillisecondsSinceEpoch(
403:                  payload['timestamp'] as int? ?? now.millisecondsSinceEpoch),
404:              createdAt: now,
405:            ));
406:        await _notify(
407:          id: int.tryParse(id) ?? 2,
408:          title: 'SMS: $address',
409:          body: body,
410:          payload: message.id,
411:        );
412:        break;
413:
414:      case MessageTypes.smsOutgoing:
415:        if (isSource) {
416:          final address = payload['address'] as String? ?? '';
417:          final body = payload['body'] as String? ?? '';
418:          final id = payload['id'] as String? ?? message.id;
419:          var status = 'sent';
420:          try {
421:            await TelephonyChannel.sendSms(address, body);
422:          } catch (e) {
423:            _logger.e('SMS send failed: $e');
424:            status = 'failed';
425:          }
426:          await _ref.read(smsListProvider.notifier).add(SmsMessage(
427:                id: id,
428:                threadId: payload['thread_id'] as String? ?? '',
429:                address: address,
430:                body: body,
431:                encrypted: message.payload,
432:                direction: 'outgoing',
433:                status: status,
434:                timestamp: DateTime.fromMillisecondsSinceEpoch(
435:                    payload['timestamp'] as int? ?? now.millisecondsSinceEpoch),
436:                createdAt: now,
437:              ));
438:          await _socketManager?.sendMessage(MessageTypes.smsStatus, {
439:            'id': id,
440:            'status': status,
441:          });
442:        }
443:        break;
444:
445:      case MessageTypes.smsStatus:
446:        final id = payload['id'] as String?;
447:        final status = payload['status'] as String? ?? 'sent';
448:        if (id != null) {
449:          await _ref.read(smsListProvider.notifier).updateStatus(id, status);
450:        }
451:        break;
452:
453:      case MessageTypes.ack:
454:        _logger.i('ACK received: ${message.id}');
455:        break;
456:
457:      default:
458:        _logger.i('Unknown message type: ${message.type}');
459:    }
460:  }

531:  Future<void> _flushQueue() async {
532:    final items = await _queue.pendingItems();
533:    if (items.isEmpty) return;
534:    _logger.i('Flushing ${items.length} queued message(s).');
535:    for (final item in items) {
536:      try {
537:        final payload = jsonDecode(item.payload) as Map<String, dynamic>;
538:        final sent = await _socketManager?.sendMessage(item.type, payload) ?? false;
539:        if (sent) {
540:          if (item.id != null) await _queue.markSent(item.id!);
541:        } else {
542:          if (item.id != null) await _queue.markFailed(item.id!, item.retryCount);
543:          break;
544:        }
545:      } catch (e) {
546:        _logger.e('Failed to flush queue item ${item.id}: $e');
547:        if (item.id != null) await _queue.markFailed(item.id!, item.retryCount);
548:      }
549:    }
550:  }
