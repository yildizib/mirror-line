CP   531	  Future<void> _flushQueue() async {
CP   532	    final items = await _queue.pendingItems();
CP   533	    if (items.isEmpty) return;
CP   534	    _logger.i('Flushing ${items.length} queued message(s).');
CP   535	    for (final item in items) {
CP   536	      try {
CP   537	        final payload = jsonDecode(item.payload) as Map<String, dynamic>;
CP   538	        final sent = await _socketManager?.sendMessage(item.type, payload) ?? false;
CP   539	        if (sent) {
CP   540	          if (item.id != null) await _queue.markSent(item.id!);
CP   541	        } else {
CP   542	          if (item.id != null) await _queue.markFailed(item.id!, item.retryCount);
CP   543	          break;
CP   544	        }
CP   545	      } catch (e) {
CP   546	        _logger.e('Failed to flush queue item ${item.id}: $e');
CP   547	        if (item.id != null) await _queue.markFailed(item.id!, item.retryCount);
CP   548	      }
CP   549	    }
CP   550	  }
