 113|  /// Reloads peer info and (re)starts role-specific networking machinery.
 114|  /// Call after pairing, role changes or resets.
 115|  Future<void> refresh() async {
 116|    try {
 117|      _peer = await _peerDao.getPeer();
 118|      _key = await KeyStore.getPeerKey();
 119|    } catch (e) {
 120|      _logger.e('Failed to load peer info: $e');
 121|      return;
 122|    }
 123|
 124|    final statusNotifier = _ref.read(connectionStatusProvider.notifier);
 125|    final localIp = await PeerDiscovery().getLocalIp();
 126|    statusNotifier.setLocalIp(localIp);
 127|    statusNotifier.setPeerIp(_peer?.ip);
 128|
 129|    if (_peer == null || _key == null) {
 130|      _logger.w('No peer info or key found. Waiting for pairing.');
 131|      await _stopMachinery();
 132|      return;
 133|    }
 134|
 135|    try {
 136|      if (isSource) {
 137|        await _startAsSource();
 138|      } else {
 139|        await _startAsMain();
 140|      }
 141|    } catch (e) {
 142|      _logger.e('Failed to start networking: $e');
 143|    }
 144|  }

 146|  Future<void> _startAsSource() async {
 147|    final peer = _peer!;
 148|    final key = _key!;
 149|
 150|    _socketManager ??= _createSocketManager();
 151|    if (_socketManager!.isConnected == false) {
 152|      try {
 153|        await _socketManager!.startServer(peer.port, key);
 154|        _ref.read(connectionStatusProvider.notifier).setServer(peer.port, true);
 155|      } catch (e) {
 156|        _logger.e('Server start failed: $e');
 157|        _ref.read(connectionStatusProvider.notifier)
 158|            .recordConnectAttempt('Server başlatılamadı: $e');
 159|      }
 160|    }
