CP   361	        final id = payload['id'] as String? ?? message.id;
CP   362	        await _ref.read(callListProvider.notifier).add(CallEvent(
CP   363	              id: id,
CP   364	              direction: 'incoming',
CP   365	              number: number,
CP   366	              timestamp: DateTime.fromMillisecondsSinceEpoch(
CP   367	                  payload['timestamp'] as int? ?? now.millisecondsSinceEpoch),
CP   368	              encrypted: message.payload,
CP   369	              status: 'delivered',
CP   370	              createdAt: now,
CP   371	            ));
CP   372	        await _notify(
CP   373	          id: int.tryParse(id) ?? 1,
CP   374	          title: 'Gelen Arama',
CP   375	          body: number,
CP   376	          payload: message.id,
CP   377	        );
CP   378	        break;
CP   379	
CP   380	      case MessageTypes.callRejected:
CP   381	        if (isSource) {
CP   382	          await TelephonyChannel.rejectCall();
CP   383	        }
CP   384	        final id = payload['id'] as String?;
CP   385	        if (id != null) {
CP   386	          await _ref.read(callListProvider.notifier).updateStatus(id, 'rejected');
CP   387	        }
CP   388	        break;
CP   389	
CP   390	      case MessageTypes.smsIncoming:
CP   391	        final address = payload['address'] as String? ?? 'unknown';
CP   392	        final body = payload['body'] as String? ?? '';
CP   393	        final id = payload['id'] as String? ?? message.id;
CP   394	        await _ref.read(smsListProvider.notifier).add(SmsMessage(
CP   395	              id: id,
CP   396	              threadId: payload['thread_id'] as String? ?? '',
CP   397	              address: address,
CP   398	              body: body,
CP   399	              encrypted: message.payload,
CP   400	              direction: 'incoming',
CP   401	              status: 'received',
CP   402	              timestamp: DateTime.fromMillisecondsSinceEpoch(
CP   403	                  payload['timestamp'] as int? ?? now.millisecondsSinceEpoch),
CP   404	              createdAt: now,
CP   405	            ));
CP   406	        await _notify(
CP   407	          id: int.tryParse(id) ?? 2,
CP   408	          title: 'SMS: $address',
CP   409	          body: body,
CP   410	          payload: message.id,
CP   411	        );
CP   412	        break;
CP   413	
CP   414	      case MessageTypes.smsOutgoing:
CP   415	        if (isSource) {
CP   416	          final address = payload['address'] as String? ?? '';
CP   417	          final body = payload['body'] as String? ?? '';
CP   418	          final id = payload['id'] as String? ?? message.id;
CP   419	          var status = 'sent';
CP   420	          try {
CP   421	            await TelephonyChannel.sendSms(address, body);
CP   422	          } catch (e) {
CP   423	            _logger.e('SMS send failed: $e');
CP   424	            status = 'failed';
CP   425	          }
CP   426	          await _ref.read(smsListProvider.notifier).add(SmsMessage(
CP   427	                id: id,
CP   428	                threadId: payload['thread_id'] as String? ?? '',
CP   429	                address: address,
CP   430	                body: body,
CP   431	                encrypted: message.payload,
CP   432	                direction: 'outgoing',
CP   433	                status: status,
CP   434	                timestamp: DateTime.fromMillisecondsSinceEpoch(
CP   435	                    payload['timestamp'] as int? ?? now.millisecondsSinceEpoch),
CP   436	                createdAt: now,
CP   437	              ));
CP   438	          await _socketManager?.sendMessage(MessageTypes.smsStatus, {
CP   439	            'id': id,
CP   440	            'status': status,
CP   441	          });
CP   442	        }
CP   443	        break;
CP   444	
CP   445	      case MessageTypes.smsStatus:
CP   446	        final id = payload['id'] as String?;
CP   447	        final status = payload['status'] as String? ?? 'sent';
CP   448	        if (id != null) {
CP   449	          await _ref.read(smsListProvider.notifier).updateStatus(id, status);
CP   450	        }
CP   451	        break;
CP   452	
CP   453	      case MessageTypes.ack:
CP   454	        _logger.i('ACK received: ${message.id}');
CP   455	        break;
CP   456	
CP   457	      default:
CP   458	        _logger.i('Unknown message type: ${message.type}');
CP   459	    }
CP   460	  }
